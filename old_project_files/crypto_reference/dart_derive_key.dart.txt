// PBKDF2-HMAC-SHA256, iterations=300000, 256-bit output, Base64URL key

import 'package:cryptography/cryptography.dart';
import 'dart:convert';

Future<String> deriveKey(String masterPassword, String saltB64Url) async {
  final pbkdf2 = Pbkdf2(
    macAlgorithm: Hmac.sha256(),
    iterations: 300000,
    bits: 256,
  );

  final saltBytes = base64Url.decode(saltB64Url);
  final secretKey = SecretKey(utf8.encode(masterPassword));
  final derivedKey = await pbkdf2.deriveKey(
    secretKey: secretKey,
    nonce: saltBytes,
  );
  final bytes = await derivedKey.extractBytes();
  return base64Url.encode(bytes); // Base64URL (no padding)
}
